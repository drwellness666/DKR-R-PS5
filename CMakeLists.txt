cmake_minimum_required(VERSION 3.24)
project(DKRPS5PlatformProbe LANGUAGES C CXX)
if(NOT PS5)
  message(FATAL_ERROR "Use the public PS5 payload SDK toolchain; this is a native PS5 build.")
endif()
set(CMAKE_C_STANDARD 11)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(LIBSAMPLERATE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_OPENGL OFF CACHE BOOL "" FORCE)
set(SDL_OPENGLES OFF CACHE BOOL "" FORCE)
set(SDL_VULKAN OFF CACHE BOOL "Diagnostic uses the software framebuffer" FORCE)
set(SDL_LIBSAMPLERATE OFF CACHE BOOL "Native output uses 48 kHz" FORCE)
if(NOT EXISTS "${PS5_SDL_SOURCE}/CMakeLists.txt" OR NOT EXISTS "${SAMPLERATE_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Set PS5_SDL_SOURCE and SAMPLERATE_SOURCE to the pinned sources.")
endif()
add_subdirectory("${SAMPLERATE_SOURCE}" samplerate EXCLUDE_FROM_ALL)
add_subdirectory("${PS5_SDL_SOURCE}" sdl EXCLUDE_FROM_ALL)
add_executable(dkrr-ps5-probe src/platform_probe.c)
set_target_properties(dkrr-ps5-probe PROPERTIES SUFFIX ".elf")
target_compile_options(dkrr-ps5-probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(dkrr-ps5-probe PRIVATE SDL2::SDL2-static)
add_executable(dkrr-ps5-probe-lowmem src/platform_probe.c)
set_target_properties(dkrr-ps5-probe-lowmem PROPERTIES SUFFIX ".elf")
target_compile_definitions(dkrr-ps5-probe-lowmem PRIVATE DKR_PROBE_BUILD_LABEL="lowmem-v2")
target_compile_options(dkrr-ps5-probe-lowmem PRIVATE -Wall -Wextra -Werror)
target_link_libraries(dkrr-ps5-probe-lowmem PRIVATE SDL2::SDL2-static)
add_executable(dkrr-ps5-cxx-probe src/cxx_probe.cpp)
set_target_properties(dkrr-ps5-cxx-probe PROPERTIES SUFFIX ".elf" CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES)
target_compile_options(dkrr-ps5-cxx-probe PRIVATE -Wall -Wextra -Werror)
add_executable(dkrr-ps5-memory-probe src/memory_probe.c)
set_target_properties(dkrr-ps5-memory-probe PROPERTIES SUFFIX ".elf")
target_compile_options(dkrr-ps5-memory-probe PRIVATE -Wall -Wextra -Werror)
